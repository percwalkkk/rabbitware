local function getClosestHand()
    local character = game.Players.LocalPlayer.Character
    local ball = game.Players.LocalPlayer.Character:FindFirstChild("Basketball")
    if not character then return nil end
    
    local rightHand = character:FindFirstChild("RightHand") or character:FindFirstChild("Right Arm")
    local leftHand = character:FindFirstChild("LeftHand") or character:FindFirstChild("Left Arm")
    
    if not rightHand or not leftHand then return nil end
    
    local ballPosition = ball.Position
    local rightDistance = (rightHand.Position - ballPosition).Magnitude
    local leftDistance = (leftHand.Position - ballPosition).Magnitude
    
    local threshold = 2
    
    if rightDistance < threshold and rightDistance < leftDistance then
        return "Right"
    elseif leftDistance < threshold then
        return "Left"
    end
    
    return nil
end

local MOVE_NAMES = {
    "Side Step",
    "Crossover",
    "Spin Move",
    "Double Crossover",
    "Stepback Switch Hands",
}

local function fireDribble(keys)
    local args = { keys }
    game:GetService("ReplicatedStorage"):WaitForChild("Events"):WaitForChild("Dribble"):FireServer(unpack(args))
end

local function executeMove(moveName)
    local hand = getClosestHand()
    if moveName == "Side Step" then
        if hand == "Left" then fireDribble({Enum.KeyCode.Z}) else fireDribble({Enum.KeyCode.C}) end
    elseif moveName == "Crossover" then
        if hand == "Left" then fireDribble({Enum.KeyCode.C})
        elseif hand == "Right" then fireDribble({Enum.KeyCode.Z})
        else fireDribble({Enum.KeyCode.X}) end
    elseif moveName == "Spin Move" then
        if hand == "Left" then fireDribble({Enum.KeyCode.C, Enum.KeyCode.C})
        else fireDribble({Enum.KeyCode.Z, Enum.KeyCode.Z}) end
    elseif moveName == "Double Crossover" then
        if hand == "Right" then fireDribble({Enum.KeyCode.C, Enum.KeyCode.C})
        else fireDribble({Enum.KeyCode.Z, Enum.KeyCode.Z}) end
    elseif moveName == "Stepback Switch Hands" then
        fireDribble({Enum.KeyCode.X, Enum.KeyCode.X})
    end
end

local MacroConnections = {}

local function stopMacro(name)
    if MacroConnections[name] then
        getgenv().SavedMacros[name].active = false
        MacroConnections[name] = false
    end
end

local function runMacro(name)
    local macro = getgenv().SavedMacros[name]
    if not macro or #macro.moves == 0 then return end
    if macro.active then return end

    macro.active = true

    coroutine.wrap(function()
        local char = game.Players.LocalPlayer.Character
        if char and char:FindFirstChild("Basketball") then
            for _, moveName in ipairs(macro.moves) do
                executeMove(moveName)
                task.wait(macro.delay or 0.1)
            end
        end
        macro.active = false
    end)()
end

local MacroKeybindConnections = {}

local function bindMacroKey(name, keyCode)
    if MacroKeybindConnections[name] then
        MacroKeybindConnections[name]:Disconnect()
        MacroKeybindConnections[name] = nil
    end
    if not keyCode then return end
    MacroKeybindConnections[name] = game:GetService("UserInputService").InputBegan:Connect(function(input, gpe)
        if gpe then return end
        if input.KeyCode == keyCode then
            runMacro(name)
        end
    end)
end

coroutine.wrap(function()

local RunService = game:GetService("RunService")
local LocalPlayer = game:GetService("Players").LocalPlayer

local function findClosestHoop()
    local courts = workspace:FindFirstChild("Courts")
    if not courts then return nil end
    if LocalPlayer.Values.InGame.Value == true then
        for _,v in pairs(workspace.Courts:GetDescendants()) do
            if v:IsA("StringValue") then
                if v.Value == LocalPlayer.Name then
                    return v.Parent.Parent.Hoop
                end
            end
        end
    end
end

local function getClosestPlayerWithBall()
    local hoop = findClosestHoop()
    if not hoop then return nil end
    local courtValues = hoop.Parent.Values
    if not courtValues then return nil end
    if not LocalPlayer.Character then return nil end
    local hrp = LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    local t1, t2 = LocalPlayer.Values.Teammate1.Value, LocalPlayer.Values.Teammate2.Value
    local myPos = hrp.Position
    local closestOpponent, shortestDistance = nil, math.huge
    for i = 1, 4 do
        local playerValueObj = courtValues:FindFirstChild("Player" .. i)
        if playerValueObj then
            local playerValue = playerValueObj.Value
            if playerValue ~= t1 and playerValue ~= t2 then
                local opponent = game.Players:FindFirstChild(playerValue)
                if opponent and opponent.Character and opponent.Character:FindFirstChild("HumanoidRootPart") and opponent.Values.Basketball ~= nil and opponent ~= LocalPlayer then
                    local distance = (myPos - opponent.Character.HumanoidRootPart.Position).Magnitude
                    if distance < shortestDistance then
                        shortestDistance = distance
                        closestOpponent = opponent
                    end
                end
            end
        end
    end
    return closestOpponent
end

RunService.Heartbeat:Connect(function()
    if not getgenv().AutoGuard then
        game:GetService("ReplicatedStorage"):WaitForChild("Events"):WaitForChild("Guard"):FireServer(false)
        return
    end
    if getgenv().AutoGuardMethod ~= "Blatant" then return end

    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    if LocalPlayer.Values.Basketball.Value ~= nil then return end
    local target = getClosestPlayerWithBall()
    if not target then game:GetService("ReplicatedStorage"):WaitForChild("Events"):WaitForChild("Guard"):FireServer(false); return end
    if not target.Values.Basketball.Value then game:GetService("ReplicatedStorage"):WaitForChild("Events"):WaitForChild("Guard"):FireServer(false); return end
    local hoop = findClosestHoop()
    if target.Character and target.Character:FindFirstChild("HumanoidRootPart") and hoop then
        local targetPos = target.Character.HumanoidRootPart.Position
        local hoopPos = hoop:GetPivot().Position
        local hoopToTarget = (targetPos - hoopPos).Unit
        local guardPos = targetPos - (hoopToTarget * (getgenv().GUARD_DISTANCE or 3))
        local lookDir = (guardPos - hoopPos).Unit
        local currentVelocity = hrp.AssemblyLinearVelocity
        hrp.CFrame = CFrame.new(guardPos, guardPos + lookDir)
        game:GetService("ReplicatedStorage"):WaitForChild("Events"):WaitForChild("Guard"):FireServer(true)
        hrp.AssemblyLinearVelocity = Vector3.new(currentVelocity.X, currentVelocity.Y, currentVelocity.Z)
    end
end)

-- Legit Auto Guard
local isMovingManually = false
local movementKeys = {[Enum.KeyCode.W]=true,[Enum.KeyCode.A]=true,[Enum.KeyCode.S]=true,[Enum.KeyCode.D]=true}
game:GetService("UserInputService").InputBegan:Connect(function(input, gpe)
    if gpe then return end
    if movementKeys[input.KeyCode] then isMovingManually = true end
end)
game:GetService("UserInputService").InputEnded:Connect(function(input)
    if movementKeys[input.KeyCode] then
        local stillMoving = false
        for key in pairs(movementKeys) do if game:GetService("UserInputService"):IsKeyDown(key) then stillMoving = true; break end end
        isMovingManually = stillMoving
    end
end)

RunService.RenderStepped:Connect(function()
    if not getgenv().AutoGuard then return end
    if getgenv().AutoGuardMethod ~= "Legit" then return end
    local myChar = LocalPlayer.Character
    if not myChar then return end
    local myHRP = myChar:FindFirstChild("HumanoidRootPart")
    local humanoid = myChar:FindFirstChildOfClass("Humanoid")
    if not myHRP or not humanoid then return end
    if LocalPlayer.Values.Basketball.Value ~= nil then return end
    if isMovingManually then return end
    local target = getClosestPlayerWithBall()
    if not target or not target.Character then return end
    local targetHRP = target.Character:FindFirstChild("HumanoidRootPart")
    if not targetHRP then return end
    local velocity = targetHRP.Velocity
    if velocity.Magnitude < 1 then velocity = targetHRP.CFrame.LookVector * 2 end
    local opponentMoveDir = Vector3.new(velocity.X, 0, velocity.Z).Unit
    local spacing = getgenv().GUARD_DISTANCE or 3
    local targetPosition = targetHRP.Position + opponentMoveDir * spacing
    local toTarget = targetPosition - myHRP.Position
    local moveDir = Vector3.new(toTarget.X, 0, toTarget.Z)
    local hoop = findClosestHoop()
    if moveDir.Magnitude > 1 then
        humanoid.WalkSpeed = getgenv().AutoGuardSpeed or 20
        humanoid:Move(moveDir.Unit, false)
        if hoop then
            local hoopPos = hoop:GetPivot().Position
            local dirToHoop = (hoopPos - myHRP.Position).Unit
            local awayFromHoop = -dirToHoop
            local lookCFrame = CFrame.new(myHRP.Position, myHRP.Position + Vector3.new(awayFromHoop.X, 0, awayFromHoop.Z))
            myHRP.CFrame = CFrame.new(myHRP.Position) * (lookCFrame - lookCFrame.Position)
        end
        game:GetService("ReplicatedStorage"):WaitForChild("Events"):WaitForChild("Guard"):FireServer(true)
    else
        humanoid:Move(Vector3.zero, false)
    end
end)
end)()

coroutine.wrap(function()
    local Players = game:GetService("Players")
    local ReplicatedStorage = game:GetService("ReplicatedStorage")
    local RunService = game:GetService("RunService")
    local player = Players.LocalPlayer
    local BlockRemote = ReplicatedStorage:WaitForChild("Events"):WaitForChild("Block")
    local currentTrackedPlayer = nil
    local currentConnection = nil
    local lastBlockTime = 0
    local BLOCK_COOLDOWN = 0.5

    local function getClosestOpponentWithBall()
        if not player.Character then return nil end
        local hrp = player.Character:FindFirstChild("HumanoidRootPart")
        if not hrp then return nil end
        local myPos = hrp.Position
        local closestOpponent, shortestDistance = nil, math.huge
        if player.Values.InGame.Value ~= true then return nil end
        local t1 = player.Values.Teammate1.Value
        local t2 = player.Values.Teammate2.Value
        local court = nil
        for _, v in pairs(workspace.Courts:GetDescendants()) do
            if v:IsA("StringValue") and v.Value == player.Name then court = v.Parent.Parent; break end
        end
        if not court then return nil end
        for i = 1, 4 do
            local playerValue = court.Values["Player" .. i].Value
            if playerValue ~= t1 and playerValue ~= t2 and playerValue ~= player.Name then
                local opponent = Players:FindFirstChild(playerValue)
                if opponent and opponent.Character then
                    local oppHrp = opponent.Character:FindFirstChild("HumanoidRootPart")
                    if oppHrp and opponent.Values.Basketball.Value ~= nil then
                        local distance = (myPos - oppHrp.Position).Magnitude
                        local MAX_DISTANCE = getgenv().AutoBlockRange or 20
                        if distance < shortestDistance and distance <= MAX_DISTANCE then
                            shortestDistance = distance; closestOpponent = opponent
                        end
                    end
                end
            end
        end
        return closestOpponent
    end

    local function performBlock()
        local currentTime = tick()
        if currentTime - lastBlockTime < BLOCK_COOLDOWN then return end
        lastBlockTime = currentTime
        BlockRemote:FireServer()
    end

    local function setupListener(targetPlayer)
        if not targetPlayer then return end
        if currentConnection then currentConnection:Disconnect(); currentConnection = nil end
        currentTrackedPlayer = targetPlayer
        local shootingValue = targetPlayer.Values:WaitForChild("Shooting", 5)
        if not shootingValue then return end
        currentConnection = shootingValue:GetPropertyChangedSignal("Value"):Connect(function()
            if shootingValue.Value == true and getgenv().AutoBlock == true then
                performBlock()
            end
        end)
    end

    task.spawn(function()
        while true do
            task.wait(0.5)
            if not getgenv().AutoBlock then
                if currentConnection then currentConnection:Disconnect(); currentConnection = nil end
                currentTrackedPlayer = nil
                task.wait(1); continue
            end
            local closestPlayer = getClosestOpponentWithBall()
            if closestPlayer ~= currentTrackedPlayer then
                if closestPlayer then setupListener(closestPlayer)
                else
                    if currentConnection then currentConnection:Disconnect(); currentConnection = nil end
                    currentTrackedPlayer = nil
                end
            end
        end
    end)

    player.CharacterAdded:Connect(function()
        task.wait(1)
        currentTrackedPlayer = nil
        if currentConnection then currentConnection:Disconnect(); currentConnection = nil end
    end)
end)()
